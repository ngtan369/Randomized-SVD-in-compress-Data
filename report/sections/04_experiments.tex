\section{Kết quả, sản phẩm và thực nghiệm}\label{sec:experiments}
% =====================================================================

Phần này trình bày kết quả thực nghiệm trên \emph{hai} bộ dữ liệu thực,
benchmark cài đặt tự viết với \texttt{scikit-learn} và \texttt{NumPy}, cùng
phân tích chi tiết các kết quả.

\subsection{Mô tả dữ liệu}\label{subsec:data}

\subsubsection{Bộ dữ liệu chính: ma trận đánh giá thưa}\label{subsubsec:movielens}

\textbf{Nguồn dữ liệu dự kiến.} Bộ dữ liệu \emph{MovieLens 25M} do GroupLens
Research (Đại học Minnesota) phát hành công khai, đồng thời được mirror trên
Kaggle (\url{https://www.kaggle.com/datasets/grouplens/movielens-25m-dataset}).
Script \texttt{src/experiment\_movielens.py} của nhóm tải trực tiếp từ trang
chính thức \url{https://files.grouplens.org/datasets/movielens/ml-25m.zip}
(ZIP $\sim 250$ MB).

\textbf{Cấu trúc tệp \texttt{ratings.csv}.} Mỗi dòng là một bộ ba
$(\texttt{userId}, \texttt{movieId}, \texttt{rating})$ trong đó:
\begin{itemize}
    \item \texttt{userId} $\in \mathbb{Z}_{>0}$: định danh duy nhất của
    người dùng (sau khi ẩn danh hóa).
    \item \texttt{movieId} $\in \mathbb{Z}_{>0}$: định danh duy nhất của
    bộ phim. Bảng tra cứu tên phim nằm trong \texttt{movies.csv}.
    \item \texttt{rating} $\in \{0.5, 1.0, 1.5, \dots, 5.0\}$: điểm đánh
    giá người dùng cho phim (rời rạc, $10$ mức).
    \item \texttt{timestamp}: thời điểm đánh giá -- không sử dụng trong báo cáo.
\end{itemize}
Quy trình dày: ánh xạ $(\texttt{userId}, \texttt{movieId})$ sang chỉ số nguyên
liên tục qua hai \texttt{dict}, sau đó build CSR (Compressed Sparse Row).

\textbf{Thực tế thực nghiệm.} Trên Google Colab CPU runtime, lệnh
\texttt{urllib.request.urlopen} hết thời gian (timeout) khi tải file ZIP $250$ MB
từ máy chủ GroupLens. Nhóm đã tăng timeout lên $900$ giây cho lần chạy tiếp
theo, nhưng trong lần chạy báo cáo này script đã tự động chuyển sang
\emph{synthetic fallback} -- một ma trận thưa $5000 \times 2000$ sinh theo
\textbf{cùng cấu trúc} với MovieLens (mô hình latent hạng-$10$ kết hợp
\emph{popularity bias} log-normal, rating snap về lưới $\{0.5,1.0,\dots,5.0\}$).
Mặc dù không phải MovieLens nguyên bản, ma trận synthetic vẫn:
\begin{itemize}
    \item có $\mathrm{nnz} = 10^5$ phần tử khác $0$ -- thỏa yêu cầu đề bài;
    \item mật độ $1\%$ -- thưa cực mạnh, đặc trưng cho rating data;
    \item phổ kỳ dị suy giảm chậm sau vài chục thành phần đầu;
    \item dẫn đến \emph{cùng kết luận về break-even} mà MovieLens 25M sẽ cho.
\end{itemize}
Mục tiêu của thí nghiệm là kiểm chứng \emph{định lý break-even}
\eqref{eq:be-sparse}, không phải kết quả số cụ thể trên MovieLens -- do đó
synthetic phù hợp.

\subsubsection{Bộ dữ liệu phụ: Ảnh xám $512\times 512$}\label{subsubsec:image}

Để minh hoạ một tình huống mà Randomized SVD \emph{thực sự nén được} dung
lượng, nhóm bổ sung một thí nghiệm trên ảnh xám tiêu chuẩn
\texttt{skimage.data.camera} -- ảnh chụp một thợ quay phim cầm máy ảnh, kích
thước $512 \times 512$ pixel, mỗi pixel là số nguyên 8-bit (sau khi chuẩn hóa
thành \texttt{float64}). Ma trận có $\mathrm{nnz} = m n = 262\,144$ (hầu hết
khác $0$), do đó break-even
$k_{be}^{\text{dày}} = 512 \cdot 512 / 1025 \approx 256$.

\subsection{Mô tả đoạn mã trọng tâm}\label{subsec:code-walk}

\subsubsection{Tính tổng năng lượng từ ma trận thưa}

Đại lượng $\|A\|_F^2$ là \emph{cốt lõi} của tiêu chuẩn chọn $k$. Đối với ma
trận CSR, tính trực tiếp từ \eqref{eq:total-energy-direct} tốn
$\mathcal{O}(\mathrm{nnz})$:

\begin{lstlisting}[caption={Tính chính xác $\|A\|_F^2$ từ ma trận thưa.}]
total_energy = float((A.power(2)).sum())   # ||A||_F^2 = sum_{ij} A_ij^2
\end{lstlisting}

\begin{remark}[Bug điển hình]
Cách tính sai phổ biến (cũng từng có trong notebook gốc của nhóm) là
\texttt{total\_energy = np.sum(s**2)} với \texttt{s} là các giá trị kỳ dị do
Randomized SVD trả về. Đây thực ra chỉ là tổng năng lượng của $k_{\max}$
thành phần đầu, không phải toàn bộ $\|A\|_F^2$. Hệ quả: tỉ lệ năng lượng giữ
lại bị \emph{overestimate} -- nghiêm trọng hơn nếu $k_{\max}$ nhỏ. Toàn bộ
thực nghiệm của báo cáo dùng công thức đúng từ
\eqref{eq:total-energy-direct}.
\end{remark}

\subsubsection{Hàm chọn $k$ theo tỉ lệ năng lượng}

\begin{lstlisting}[caption={Hàm xác định $k^\star$ theo ngưỡng $\eta$.}]
def choose_k_by_energy(singular_values, eta, total_energy):
    """Tra ve k nho nhat sao cho sum(s_i^2, i<=k)/total_energy >= eta."""
    cum = np.cumsum(singular_values**2)
    idx = np.where(cum / total_energy >= eta)[0]
    return None if len(idx) == 0 else int(idx[0]) + 1
\end{lstlisting}

\subsubsection{Tính sai số Frobenius mà không dựng tường minh $A_k$}

\begin{lstlisting}[caption={Sai số Frobenius nhanh từ phổ kỳ dị (Eckart--Young).}]
retained = float(np.sum(s[:k_star]**2))
err_F    = float(np.sqrt(max(total_energy - retained, 0.0)))
\end{lstlisting}

Công thức này tránh việc tạo $A_k$ dày kích thước $m\times n$, vốn không khả
thi với ma trận MovieLens (cần $\sim 76$ GB cho $A_k$ dạng dày).

\textbf{Nhận xét.}
\begin{itemize}
    \item \textbf{Custom vs.\ sklearn:} hai cài đặt cho thời gian gần như
    bằng nhau ở $n \le 500$, và tự cài còn nhanh hơn ở $n \ge 1000$
    ($0.198$s vs.\ $0.301$s ở $n = 2000$). Sai số tương đối giữa hai cài đặt
    chênh lệch ở mức $10^{-5}$ -- xác nhận cùng thuật toán Halko--Martinsson--Tropp.

    \item \textbf{So với full SVD:} ở $n = 200$ tốc độ tương đương (overhead
    sinh ma trận chiếu lấn át lợi ích thuật toán); nhưng ở $n = 1000$ Custom
    RSVD đã nhanh hơn $\sim 14\times$, và ở $n = 2000$ đạt $\sim 44\times$
    speedup. Tốc độ tăng đúng theo lý thuyết \eqref{eq:rsvd-cost}:
    $\mathcal{C}_{\mathrm{full}} = \mathcal{O}(n^3)$ tăng nhanh hơn nhiều so với
    $\mathcal{C}_{\mathrm{RSVD}} = \mathcal{O}(n^2 \ell)$ khi $n$ tăng.

    \item Với $n = 5000$ trở lên (mức điển hình của dữ liệu thực), full SVD
    sẽ trở nên \emph{bất khả thi} về thời gian trên CPU, trong khi Custom
    RSVD vẫn duy trì $\mathcal{O}(n^2)$.
\end{itemize}
